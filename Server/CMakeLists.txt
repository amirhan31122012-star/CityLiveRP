cmake_minimum_required(VERSION 3.18)
project(CityLiveSamp LANGUAGES C CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
 message(FATAL_ERROR "CityLive requires Linux i386. Use the Dockerfile or configure both CMAKE_C_FLAGS and CMAKE_CXX_FLAGS with -m32.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(SAMPGDK_STATIC ON CACHE BOOL "" FORCE)
set(SAMPGDK_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(SAMPSDK_DIR "${CMAKE_CURRENT_SOURCE_DIR}/ThirdParty/samp-sdk" CACHE PATH "" FORCE)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
set(PYTHON_EXECUTABLE /usr/bin/python3 CACHE FILEPATH "" FORCE)
add_subdirectory(ThirdParty/sampgdk)
file(GLOB_RECURSE ENGINE_SOURCES CONFIGURE_DEPENDS ThirdParty/samp-cpp/Engine/src/*.cpp)
add_library(citylive SHARED ${ENGINE_SOURCES} ThirdParty/samp-cpp/ServerCore/src/ServerCore.cpp ThirdParty/samp-sdk/amxplugin.cpp src/GameMode.cpp)
target_include_directories(citylive PRIVATE ThirdParty/samp-cpp/Engine/include ThirdParty/quickmaffs/include ThirdParty/samp-sdk ThirdParty/samp-sdk/amx ThirdParty/sampgdk/include "${CMAKE_CURRENT_BINARY_DIR}/ThirdParty/sampgdk/include")
target_compile_definitions(citylive PRIVATE LINUX QUICKMAFFS_HEADER_ONLY SAMPCPP_PCH="SAMPCpp/EnginePCH.hpp")
target_link_libraries(citylive PRIVATE sampgdk dl pthread)
set_target_properties(citylive PROPERTIES PREFIX "" OUTPUT_NAME "citylive")

set_source_files_properties(ThirdParty/samp-cpp/ServerCore/src/ServerCore.cpp PROPERTIES COMPILE_DEFINITIONS "OnDialogResponse=CityFrameworkOnDialogResponse")
set(WITH_UNIT_TESTS OFF CACHE BOOL "" FORCE)
set(WITH_EXTERNAL_ZLIB ON CACHE BOOL "" FORCE)
set(WITH_SSL OPENSSL CACHE STRING "" FORCE)
set(WITH_CURL OFF CACHE BOOL "" FORCE)
set(INSTALL_PLUGINDIR "lib/mariadb/plugin" CACHE STRING "" FORCE)
set(CITY_OPENSSL_ROOT "" CACHE PATH "Optional source-built static OpenSSL prefix")
if(CITY_OPENSSL_ROOT)
 set(OPENSSL_INCLUDE_DIR "${CITY_OPENSSL_ROOT}/include" CACHE PATH "" FORCE)
 set(OPENSSL_CRYPTO_LIBRARY "${CITY_OPENSSL_ROOT}/lib/libcrypto.a" CACHE FILEPATH "" FORCE)
 set(OPENSSL_SSL_LIBRARY "${CITY_OPENSSL_ROOT}/lib/libssl.a" CACHE FILEPATH "" FORCE)
else()
 set(OPENSSL_INCLUDE_DIR /usr/include CACHE PATH "" FORCE)
 set(OPENSSL_CRYPTO_LIBRARY /usr/lib/i386-linux-gnu/libcrypto.a CACHE FILEPATH "" FORCE)
 set(OPENSSL_SSL_LIBRARY /usr/lib/i386-linux-gnu/libssl.a CACHE FILEPATH "" FORCE)
endif()
set(ZLIB_LIBRARY /usr/lib/i386-linux-gnu/libz.a CACHE FILEPATH "" FORCE)
set(CLIENT_PLUGIN_CACHING_SHA2_PASSWORD STATIC CACHE STRING "" FORCE)
set(CLIENT_PLUGIN_SHA256_PASSWORD STATIC CACHE STRING "" FORCE)
add_subdirectory(ThirdParty/mariadb-connector)
target_include_directories(citylive PRIVATE ThirdParty/mariadb-connector/include "${CMAKE_CURRENT_BINARY_DIR}/ThirdParty/mariadb-connector/include")
target_include_directories(citylive PRIVATE "${OPENSSL_INCLUDE_DIR}")
target_link_libraries(citylive PRIVATE mariadbclient)
# Use the same exact archives for game code and Connector/C. Do not retain
# find_library cache entries from a different OpenSSL prefix/toolchain.
set(CITY_ssl_LIB "${OPENSSL_SSL_LIBRARY}")
set(CITY_crypto_LIB "${OPENSSL_CRYPTO_LIBRARY}")
set(CITY_z_LIB "${ZLIB_LIBRARY}")
target_link_libraries(citylive PRIVATE ${CITY_ssl_LIB} ${CITY_crypto_LIB} ${CITY_z_LIB})
target_link_options(citylive PRIVATE -static-libstdc++ -static-libgcc -Wl,-z,relro,-z,now)

target_link_options(citylive PRIVATE "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map")
enable_testing()
add_executable(city_rules tests/RulesTest.cpp)
target_include_directories(city_rules PRIVATE src)
add_test(NAME city_rules COMMAND city_rules)
add_executable(city_events tests/EventTest.cpp)
target_include_directories(city_events PRIVATE tests ThirdParty/samp-cpp/Engine/include)
target_compile_definitions(city_events PRIVATE SAMPCPP_PCH="EventTestPCH.hpp" _GLIBCXX_DEBUG)
target_compile_options(city_events PRIVATE -UNDEBUG)
add_test(NAME city_events COMMAND city_events)
add_executable(city_plugin_load tests/PluginLoadTest.c)
target_link_libraries(city_plugin_load PRIVATE dl)
add_dependencies(city_plugin_load citylive)
add_test(NAME city_plugin_load COMMAND city_plugin_load $<TARGET_FILE:citylive>)

add_executable(city_database tests/DatabaseTest.cpp)
target_include_directories(city_database PRIVATE src ThirdParty/mariadb-connector/include "${CMAKE_CURRENT_BINARY_DIR}/ThirdParty/mariadb-connector/include")
target_include_directories(city_database PRIVATE "${OPENSSL_INCLUDE_DIR}")
target_link_libraries(city_database PRIVATE mariadbclient ${CITY_ssl_LIB} ${CITY_crypto_LIB} ${CITY_z_LIB} dl pthread)
target_link_options(city_database PRIVATE -static-libstdc++ -static-libgcc)

target_compile_options(city_rules PRIVATE -UNDEBUG)
target_compile_options(city_database PRIVATE -UNDEBUG)
